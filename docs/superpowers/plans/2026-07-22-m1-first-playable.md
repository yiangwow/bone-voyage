# Bone Voyage M1 (First Playable) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A playable Godot build where cheerful skeletons auto-march at one kingdom, die individually on screen while an aggregate sim drives exact numbers, and the player spends bones in a 3-item shop.

**Architecture:** Pure C# sim (`sim/BoneVoyage.Sim`, zero Godot deps, xUnit-tested) computes everything per fixed 0.1s tick and emits typed events. Godot 4.4.1 .NET project (`game/`) renders individual skeleton actors that consume those events; views never compute gameplay. Pixel art is generated by a Python/Pillow pipeline in `tools/spritegen/`.

**Tech Stack:** Godot 4.4.1-stable (.NET), C# net8.0, xUnit, Python 3 + Pillow.

## Global Constraints

- Godot version: **4.4.1-stable** (Godot.NET.Sdk/4.4.1); sim + tests target **net8.0**.
- Git author for every commit: the owner's git identity. **No Co-Authored-By lines.**
- All gameplay math lives in `BoneVoyage.Sim`. Godot scripts may only read `Simulation`/`SimState` and consume events. If a view file computes a gameplay number, that is a bug.
- All player-facing UI strings go through Godot `Tr()` with the English text as the key (localization prep, M4 finishes it).
- Sim is deterministic: no RNG, no wall-clock reads inside `BoneVoyage.Sim`. Cosmetic randomness (death-animation variant choice, spawn jitter) is allowed in `game/` only.
- Balance numbers live only in `game/balance.json` (single source; tests consume a build-time copy). No tuning constants hard-coded in C#.
- Tone rule for copy: skeletons are cheerful union workers; UI copy plays it straight (e.g. "employees recycled"). Never grim.

using System.Collections.Generic;
using System.Text.Json;
using System.Text.Json.Serialization;

namespace BoneVoyage.Sim;

public sealed record SkeletonBalance(double Speed, double Hp, double WallDamage, double Bounty);
public sealed record TowerBalance(string Name, int Zone, double Dps);
public sealed record KingdomBalance(double BaseWallHp, double WallHpGrowth, double DpsGrowth,
    double BaseJackpot, double JackpotGrowth);
public sealed record UpgradeBalance(double BaseValue, double PerLevel, double BaseCost, double CostGrowth);

public sealed record BalanceData(
    double TickSeconds,
    int ZoneCount,
    double ZoneLength,
    Dictionary<SkeletonTypeId, SkeletonBalance> Skeletons,
    List<TowerBalance> Towers,
    KingdomBalance Kingdom,
    Dictionary<UpgradeId, UpgradeBalance> Upgrades,
    double StartingBones)
{
    private static readonly JsonSerializerOptions Options = new()
    {
        PropertyNameCaseInsensitive = true,
        Converters = { new JsonStringEnumConverter() }
    };

    public static BalanceData Load(string json) =>
        JsonSerializer.Deserialize<BalanceData>(json, Options)
        ?? throw new JsonException("balance.json deserialized to null");
}

`sim/BoneVoyage.Sim/Simulation.cs` (spawning + upgrade math now; `Move`/`TowerFire` land in Tasks 4–5):

```csharp
using System;
using System.Collections.Generic;

namespace BoneVoyage.Sim;

public sealed class Simulation
{
    public BalanceData Balance { get; }
    public SimState State { get; }

    internal static readonly SkeletonTypeId[] Types =
        { SkeletonTypeId.Marcher, SkeletonTypeId.Shieldbearer };

    public Simulation(BalanceData balance)
    {
        Balance = balance;
        State = new SimState(Types.Length, balance.ZoneCount);
        State.Bones = balance.StartingBones;
        State.WallHp = WallHpForTier(1);
    }

    public double GetUpgradeValue(UpgradeId id)
    {
        var u = Balance.Upgrades[id];
        return u.BaseValue + u.PerLevel * State.UpgradeLevels.GetValueOrDefault(id);
    }

    public double GetUpgradeCost(UpgradeId id)
    {
        var u = Balance.Upgrades[id];
        return u.BaseCost * Math.Pow(u.CostGrowth, State.UpgradeLevels.GetValueOrDefault(id));
    }

    public bool TryPurchase(UpgradeId id)
    {
        var cost = GetUpgradeCost(id);
        if (State.Bones < cost) return false;
        State.Bones -= cost;
        State.UpgradeLevels[id] = State.UpgradeLevels.GetValueOrDefault(id) + 1;
        return true;
    }

    public double WallHpForTier(int tier) =>
        Balance.Kingdom.BaseWallHp * Math.Pow(Balance.Kingdom.WallHpGrowth, tier - 1);

    public List<ISimEvent> Tick()
    {
        var events = new List<ISimEvent>();
        double dt = Balance.TickSeconds;
        Spawn(events, dt);
        return events;
    }

    private void Spawn(List<ISimEvent> events, double dt)
    {
        SpawnType(SkeletonTypeId.Marcher, UpgradeId.MarcherRate, events, dt);
        SpawnType(SkeletonTypeId.Shieldbearer, UpgradeId.ShieldbearerRate, events, dt);
    }

    private void SpawnType(SkeletonTypeId type, UpgradeId rateUpgrade,
        List<ISimEvent> events, double dt)
    {
        int t = (int)type;
        State.SpawnAcc[t] += GetUpgradeValue(rateUpgrade) * dt;
        int n = (int)State.SpawnAcc[t];
        if (n <= 0) return;
        State.SpawnAcc[t] -= n;
        State.Counts[t, 0] += n;
        events.Add(new SkeletonsSpawned(type, n));
    }
}
```

using Xunit;

namespace BoneVoyage.Sim.Tests;

public class WallTests
{
    private static Simulation NoTowerSim()
    {
        var b = TestBalance.Make();
        var sim = new Simulation(b with { Towers = new() });
        return sim;
    }

    [Fact]
    public void WallHitDealsDamagePaysBountyAndKills()
    {
        var sim = NoTowerSim();
        int t = (int)SkeletonTypeId.Marcher;
        int last = sim.Balance.ZoneCount - 1;
        sim.State.Counts[t, last] = 1;
        sim.State.MoveAcc[t, last] = 0.999; // arrives at wall within a tick or two
        double wallBefore = sim.State.WallHp;
        WallHit? hit = null;
        for (int i = 0; i < 40 && hit == null; i++)
            hit = sim.Tick().OfType<WallHit>().FirstOrDefault();
        Assert.NotNull(hit);
        Assert.Equal(5.0, hit!.Damage);                    // marcher wallDamage 5
        Assert.Equal(wallBefore - 5.0, sim.State.WallHp);
        Assert.True(sim.State.Bones >= 1.0);               // bounty 1 * recovery 1.0
    }

    [Fact]
    public void KingdomFallsPaysJackpotAndScalesTier()
    {
        var sim = NoTowerSim();

        sim.State.WallHp = 4.0; // one marcher hit (5 dmg) fells it
        sim.State.Counts[t, last] = 1;
        sim.State.MoveAcc[t, last] = 0.999;

```csharp
    private void Move(List<ISimEvent> events, double dt)
    {
        foreach (var type in Types)
        {
            int t = (int)type;
            double transit = Balance.ZoneLength / Balance.Skeletons[type].Speed;
            for (int z = Balance.ZoneCount - 1; z >= 0; z--)
            {
                State.MoveAcc[t, z] += State.Counts[t, z] * dt / transit;
                int k = Math.Min((int)State.MoveAcc[t, z], State.Counts[t, z]);
                if (k <= 0) continue;
                State.MoveAcc[t, z] -= k;
                State.Counts[t, z] -= k;
                if (z == Balance.ZoneCount - 1) HitWall(type, k, events);
                else State.Counts[t, z + 1] += k;
            }
        }
    }

    private void HitWall(SkeletonTypeId type, int count, List<ISimEvent> events)
    {
        var sk = Balance.Skeletons[type];
        double damage = count * sk.WallDamage;
        State.WallHp -= damage;
        events.Add(new WallHit(type, count, damage));
        AwardBounty(type, count);
        events.Add(new SkeletonsDied(type, count, Balance.ZoneCount - 1, DeathCause.WallCrumble));
        while (State.WallHp <= 0)
        {
            double jackpot = Balance.Kingdom.BaseJackpot
                * Math.Pow(Balance.Kingdom.JackpotGrowth, State.KingdomTier - 1);
            State.Bones += jackpot;
            events.Add(new KingdomFell(State.KingdomTier, jackpot));
            State.KingdomTier++;
            State.WallHp += WallHpForTier(State.KingdomTier);
        }
    }

    private void AwardBounty(SkeletonTypeId type, int count) =>
        State.Bones += count * Balance.Skeletons[type].Bounty * GetUpgradeValue(UpgradeId.Recovery);
```

    private void TowerFire(List<ISimEvent> events, double dt)
    {
        double dpsMult = Math.Pow(Balance.Kingdom.DpsGrowth, State.KingdomTier - 1);
        foreach (var tower in Balance.Towers)
        {
            int z = tower.Zone;
            int total = 0;
            foreach (var type in Types) total += State.Counts[(int)type, z];
            if (total == 0) { State.DamagePool[z] = 0; continue; }

            State.DamagePool[z] += tower.Dps * dpsMult * dt;
            KillInZone(SkeletonTypeId.Shieldbearer, z, events);
            if (State.Counts[(int)SkeletonTypeId.Shieldbearer, z] == 0)
                KillInZone(SkeletonTypeId.Marcher, z, events);
        }
    }

    private void KillInZone(SkeletonTypeId type, int z, List<ISimEvent> events)
    {
        int t = (int)type;
        if (State.Counts[t, z] == 0) return;
        double hp = Balance.Skeletons[type].Hp;
        int kills = Math.Min(State.Counts[t, z], (int)(State.DamagePool[z] / hp));
        if (kills <= 0) return;
        State.DamagePool[z] -= kills * hp;
        State.Counts[t, z] -= kills;
        AwardBounty(type, kills);
        events.Add(new SkeletonsDied(type, kills, z, DeathCause.Tower));
    }

from PIL import Image, ImageDraw
import palette as P

F = 24  # frame size

def _frame():
    img = Image.new("RGBA", (F, F), P.TRANSPARENT)
    return img, ImageDraw.Draw(img)

def _head(d, cx, cy, front=False):
    d.ellipse([cx - 3, cy - 3, cx + 3, cy + 3], fill=P.BONE, outline=P.OUTLINE)
    if front:
        d.point([(cx - 2, cy), (cx + 2, cy)], fill=P.EYE)
        d.point([(cx - 1, cy + 2), (cx + 1, cy + 2)], fill=P.EYE)  # little smile
    else:
        d.point([(cx + 2, cy)], fill=P.EYE)  # facing right

def _torso(d, cx, top, bottom):
    d.line([(cx, top), (cx, bottom)], fill=P.BONE)          # spine
    for i, y in enumerate(range(top + 1, bottom - 1, 2)):    # ribs
        w = 2 if i < 2 else 1
        d.line([(cx - w, y), (cx + w, y)], fill=P.BONE_SHADE)

def _limb(d, x0, y0, x1, y1):
    d.line([(x0, y0), (x1, y1)], fill=P.BONE)

def walk_frame(i, shield=False):
    """4-frame walk cycle, marching right. Leg poses cycle, slight head bob."""
    img, d = _frame()
    bob = [0, -1, 0, -1][i]
    cx, hip = 11, 16 + bob
    _head(d, cx + 1, 6 + bob)
    _torso(d, cx, 10 + bob, hip)
    legs = [((cx - 3, 21), (cx + 3, 20)), ((cx - 1, 21), (cx + 1, 21)),
            ((cx + 3, 20), (cx - 3, 21)), ((cx - 1, 21), (cx + 1, 21))][i]
    _limb(d, cx, hip, *legs[0]); _limb(d, cx, hip, *legs[1])
    arm = [(cx + 4, 14), (cx + 3, 13), (cx - 2, 14), (cx + 3, 13)][i]
    _limb(d, cx, 11 + bob, *arm)
    if shield:
        d.rectangle([cx + 4, 9 + bob, cx + 8, 17 + bob], fill=P.SHIELD, outline=P.SHIELD_RIM)
        d.line([(cx - 4, 5 + bob), (cx + 5, 5 + bob)], fill=P.HELMET)  # helmet brim
    return img

def _bone_pile(d, stage):
    """stage 0: slumping; 1: tall pile; 2: flat pile + sparkles."""
    if stage == 0:
        d.ellipse([7, 14, 15, 20], outline=P.OUTLINE, fill=P.BONE_SHADE)
        _head(d, 11, 12, front=True)
    elif stage == 1:
        d.ellipse([6, 17, 17, 21], fill=P.BONE_SHADE, outline=P.OUTLINE)
        _head(d, 11, 16, front=True)
        d.point([(11 - 3, 16 + 1), (11 + 3, 16 + 1)], fill=P.BLUSH)
    else:
        d.ellipse([5, 19, 18, 22], fill=P.BONE_SHADE, outline=P.OUTLINE)
        d.point([(4, 14), (18, 12), (10, 10), (20, 17)], fill=P.SPARK)

def die_frame(i, variant, shield=False):
    """6 frames: 0 face camera, 1-2 gesture (wave or salute), 3-5 cheerful crumble."""
    img, d = _frame()
    cx = 11
    if i <= 2:
        _head(d, cx, 6, front=True)
        d.point([(cx - 3, 7), (cx + 3, 7)], fill=P.BLUSH)
        _torso(d, cx, 10, 16)
        _limb(d, cx, 16, cx - 2, 21); _limb(d, cx, 16, cx + 2, 21)
        if variant == "wave":
            tips = [(cx + 5, 8), (cx + 6, 4), (cx + 4, 3)]  # arm raising & waving
            _limb(d, cx, 11, *tips[i])
            _limb(d, cx, 12, cx - 4, 15)
        else:  # salute
            if i == 0: _limb(d, cx, 11, cx + 5, 12)
            else:      _limb(d, cx, 11, cx + 3, 6)  # crisp salute to the brow
            _limb(d, cx, 12, cx - 4, 15)
        if shield and i == 0:
            d.rectangle([cx - 8, 12, cx - 4, 20], fill=P.SHIELD, outline=P.SHIELD_RIM)
    else:
        _bone_pile(d, i - 3)
        if shield and i == 3:
            d.rectangle([3, 17, 7, 21], fill=P.SHIELD, outline=P.SHIELD_RIM)
    return img

def whack_frame(i, shield=False):
    """6 frames: 0 windup, 1 swing, 2 impact stars, 3-5 crumble."""
    img, d = _frame()
    cx = 11
    if i == 0:
        _head(d, cx - 1, 6); _torso(d, cx - 1, 10, 16)
        _limb(d, cx - 1, 11, cx - 6, 6); _limb(d, cx - 1, 11, cx - 5, 8)
        _limb(d, cx - 1, 16, cx - 4, 21); _limb(d, cx - 1, 16, cx + 1, 21)
    elif i == 1:
        _head(d, cx + 1, 6); _torso(d, cx + 1, 10, 16)
        _limb(d, cx + 1, 11, cx + 7, 9); _limb(d, cx + 1, 11, cx + 6, 11)
        _limb(d, cx + 1, 16, cx - 2, 21); _limb(d, cx + 1, 16, cx + 3, 21)
    elif i == 2:
        _head(d, cx + 2, 7); _torso(d, cx + 2, 11, 16)
        _limb(d, cx + 2, 12, cx + 8, 10)
        d.point([(cx + 9, 6), (cx + 11, 9), (cx + 10, 13), (cx + 8, 4)], fill=P.SPARK)
        _limb(d, cx + 2, 16, cx - 1, 21); _limb(d, cx + 2, 16, cx + 4, 21)
    else:
        _bone_pile(d, i - 3)
        d.point([(19, 8), (21, 12)], fill=P.SPARK)
    return img

ANIMS = {
    "walk":       {"row": 0, "frames": 4, "fps": 8,  "loop": True,  "fn": walk_frame},
    "die_wave":   {"row": 1, "frames": 6, "fps": 9,  "loop": False, "fn": lambda i, s: die_frame(i, "wave", s)},
    "die_salute": {"row": 2, "frames": 6, "fps": 9,  "loop": False, "fn": lambda i, s: die_frame(i, "salute", s)},
    "whack":      {"row": 3, "frames": 6, "fps": 10, "loop": False, "fn": whack_frame},
}

def build_sheet(shield=False):
    cols = max(a["frames"] for a in ANIMS.values())
    sheet = Image.new("RGBA", (cols * F, len(ANIMS) * F), P.TRANSPARENT)
    for a in ANIMS.values():
        for i in range(a["frames"]):
            frame = a["fn"](i, shield) if a["fn"] is not walk_frame else walk_frame(i, shield)
            sheet.paste(frame, (i * F, a["row"] * F))
    meta = {"frameSize": F, "animations": {
        k: {kk: v[kk] for kk in ("row", "frames", "fps", "loop")} for k, v in ANIMS.items()}}
    return sheet, meta
```

namespace BoneVoyage.Game;

public static class Fmt
{
    public static string Num(double v)
    {
        if (v < 1000) return ((long)v).ToString();
        string[] suffix = { "K", "M", "B", "T" };
        int i = -1;
        while (v >= 1000 && i < suffix.Length - 1) { v /= 1000; i++; }
        return $"{v:0.#}{suffix[i]}";
    }
}

    }

    public override void _Process(double delta)
    {
        _acc += delta;
        double dt = _sim.Balance.TickSeconds;
        int safety = 0;
        while (_acc >= dt && safety++ < 120)
        {
            _acc -= dt;
            List<ISimEvent> events = _sim.Tick();
            _battlefield?.OnSimEvents(events);
        }
        _hud.Refresh();

        if (_shotCountdown > 0 && --_shotCountdown == 0)
        {
            GetViewport().GetTexture().GetImage().SavePng(_shotPath);
            GetTree().Quit();
        }
    }
}

namespace BoneVoyage.Game;

public partial class Hud : CanvasLayer

    private readonly Simulation _sim;
    private Label _bones = null!, _tier = null!, _employees = null!;
    private ProgressBar _wallBar = null!;

    public Hud(Simulation sim) => _sim = sim;

    public override void _Ready()
    {
        _bones = MakeLabel(new Vector2(16, 12), 28);
        _tier = MakeLabel(new Vector2(16, 48), 18);
        _employees = MakeLabel(new Vector2(16, 74), 14);

        _wallBar = new ProgressBar
        {
            Position = new Vector2(980, 16), Size = new Vector2(284, 22),
            ShowPercentage = false, MaxValue = 1.0
        };
        AddChild(_wallBar);
    }

    private Label MakeLabel(Vector2 pos, int size)
    {
        var l = new Label { Position = pos };
        l.AddThemeFontSizeOverride("font_size", size);
        AddChild(l);
        return l;
    }

    public void Refresh()
    {
        _bones.Text = $"{Tr("Bones")}: {Fmt.Num(_sim.State.Bones)}";
        _tier.Text = $"{Tr("Meadowshire")} {ToRoman(_sim.State.KingdomTier)}";
        int alive = 0;
        for (int t = 0; t < 2; t++)
            for (int z = 0; z < _sim.Balance.ZoneCount; z++)
                alive += _sim.State.Counts[t, z];
        _employees.Text = $"{Tr("Employees on shift")}: {Fmt.Num(alive)}";
        _wallBar.Value = _sim.State.WallHp / _sim.WallHpForTier(_sim.State.KingdomTier);
    }

    private static string ToRoman(int n)
    {
        if (n >= 40) return n.ToString(); // romans get silly; switch to digits
        string[] tens = { "", "X", "XX", "XXX" };
        string[] ones = { "", "I", "II", "III", "IV", "V", "VI", "VII", "VIII", "IX" };
        return tens[n / 10] + ones[n % 10];
    }
}

using Godot;

namespace BoneVoyage.Game;

public static class SpriteSheetLoader
{
    public static SpriteFrames Load(string basePath)
    {
        var tex = GD.Load<Texture2D>($"{basePath}.png");
        var meta = Json.ParseString(FileAccess.GetFileAsString($"{basePath}.meta.json"))
            .AsGodotDictionary();
        int fs = (int)meta["frameSize"].AsDouble();
        var frames = new SpriteFrames();
        frames.RemoveAnimation("default");
        var anims = meta["animations"].AsGodotDictionary();
        foreach (var key in anims.Keys)
        {
            string name = key.AsString();
            var def = anims[key].AsGodotDictionary();
            int row = (int)def["row"].AsDouble();
            int count = (int)def["frames"].AsDouble();
            frames.AddAnimation(name);
            frames.SetAnimationSpeed(name, def["fps"].AsDouble());
            frames.SetAnimationLoop(name, def["loop"].AsBool());
            for (int i = 0; i < count; i++)
            {
                var at = new AtlasTexture
                {
                    Atlas = tex,
                    Region = new Rect2(i * fs, row * fs, fs, fs)
                };
                frames.AddFrame(name, at);
            }
        }
        return frames;
    }
}

using BoneVoyage.Sim;
using Godot;

namespace BoneVoyage.Game;

public partial class SkeletonActor : Node2D
{
    public SkeletonTypeId Type;
    public bool Walking { get; private set; } = true;
    private AnimatedSprite2D _sprite = null!;
    private float _speedPx;
    private float _stopX;

    public SkeletonActor(SkeletonTypeId type, SpriteFrames frames, float speedPx, float stopX)
    {
        Type = type;
        _speedPx = speedPx;
        _stopX = stopX;
        _sprite = new AnimatedSprite2D { SpriteFrames = frames, Scale = new Vector2(2, 2) };
        AddChild(_sprite);
        _sprite.Play("walk");
        _sprite.AnimationFinished += OnAnimFinished;
    }

    public override void _Process(double delta)
    {
        if (!Walking) return;
        var x = Position.X + _speedPx * (float)delta;
        Position = new Vector2(Mathf.Min(x, _stopX), Position.Y);
    }

    public void PlayDeath(string anim) // "die_wave" | "die_salute" | "whack"
    {
        if (!Walking) return;
        Walking = false;
        _sprite.Play(anim);
    }

    private void OnAnimFinished()
    {
        if (!Walking) QueueFree();
    }
}

using System.Collections.Generic;
using System.Linq;
using BoneVoyage.Sim;
using Godot;

namespace BoneVoyage.Game;

public partial class BattlefieldView : Node2D
{
    public const float RoadStartX = 140f, WallX = 1140f, GroundY = 430f, PxPerUnit = 10f;
    private const int ActorCap = 150;

    private readonly Simulation _sim;
    private readonly List<SkeletonActor> _actors = new();
    private readonly Dictionary<SkeletonTypeId, SpriteFrames> _frames = new();
    private static readonly string[] DeathAnims = { "die_wave", "die_salute" };

    public BattlefieldView(Simulation sim) => _sim = sim;

    public override void _Ready()
    {
        AddChild(new ColorRect { Color = new Color("b2d6eb"), Size = new Vector2(1280, GroundY), ZIndex = -20 });
        AddChild(new ColorRect { Color = new Color("76b262"), Position = new Vector2(0, GroundY), Size = new Vector2(1280, 720 - GroundY), ZIndex = -20 });
        AddChild(new ColorRect { Color = new Color("5e9450"), Position = new Vector2(0, GroundY + 20), Size = new Vector2(1280, 26), ZIndex = -19 }); // road strip

        _frames[SkeletonTypeId.Marcher] = SpriteSheetLoader.Load("res://assets/sprites/marcher");
        _frames[SkeletonTypeId.Shieldbearer] = SpriteSheetLoader.Load("res://assets/sprites/shieldbearer");

        Place("res://assets/sprites/necropolis.png", new Vector2(40, GroundY - 108), 2);
        Place("res://assets/sprites/wall.png", new Vector2(WallX, GroundY - 300), 2);
        foreach (var tower in _sim.Balance.Towers)
        {
            float x = RoadStartX + (tower.Zone + 0.5f) * (float)_sim.Balance.ZoneLength * PxPerUnit;
            string sprite = tower.Name == "Cannon" ? "tower_cannon" : "tower_arrow";
            Place($"res://assets/sprites/{sprite}.png", new Vector2(x - 24, GroundY - 140), 2);
        }
    }

    private void Place(string path, Vector2 pos, float scale)
    {
        AddChild(new Sprite2D
        {
            Texture = GD.Load<Texture2D>(path), Position = pos,
            Scale = new Vector2(scale, scale), Centered = false, ZIndex = -10
        });
    }

    public void OnSimEvents(List<ISimEvent> events)
    {
        foreach (var e in events)
        {
            switch (e)
            {
                case SkeletonsSpawned s: SpawnActors(s); break;
                case SkeletonsDied d when d.Cause == DeathCause.Tower: KillActors(d); break;
                case WallHit w: WhackActors(w); break;
                // WallCrumble deaths are covered by the whack animation; KingdomFell is HUD-only in M1
            }
        }
    }

    private void SpawnActors(SkeletonsSpawned e)
    {
        int alive = _actors.Count(a => IsInstanceValid(a));
        _actors.RemoveAll(a => !IsInstanceValid(a));
        for (int i = 0; i < e.Count && alive + i < ActorCap; i++)
        {
            float speedPx = (float)_sim.Balance.Skeletons[e.Type].Speed * PxPerUnit;
            var actor = new SkeletonActor(e.Type, _frames[e.Type], speedPx, WallX - 20f)
            {
                Position = new Vector2(RoadStartX, GroundY + (float)GD.RandRange(-6, 18)),
            };
            actor.ZIndex = (int)actor.Position.Y; // painter's order by depth
            _actors.Add(actor);
            AddChild(actor);
        }
    }

    private void KillActors(SkeletonsDied e)
    {
        float centerX = RoadStartX + (e.Zone + 0.5f) * (float)_sim.Balance.ZoneLength * PxPerUnit;
        var victims = _actors
            .Where(a => IsInstanceValid(a) && a.Walking && a.Type == e.Type)
            .OrderBy(a => Mathf.Abs(a.Position.X - centerX))
            .Take(e.Count);
        foreach (var v in victims)
            v.PlayDeath(DeathAnims[GD.RandRange(0, DeathAnims.Length - 1)]);
    }

    private void WhackActors(WallHit e)
    {
        var victims = _actors
            .Where(a => IsInstanceValid(a) && a.Walking && a.Type == e.Type)
            .OrderByDescending(a => a.Position.X)
            .Take(e.Count);
        foreach (var v in victims)
            v.PlayDeath("whack");
    }
}
```

using System.Collections.Generic;
using BoneVoyage.Sim;
using Godot;

namespace BoneVoyage.Game;

public partial class ShopPanel : PanelContainer
{
    private readonly Simulation _sim;
    private readonly List<(UpgradeId Id, string Name, string Effect, Button Btn)> _rows = new();

    public ShopPanel(Simulation sim) => _sim = sim;

    public override void _Ready()
    {
        Position = new Vector2(1000, 60);
        Size = new Vector2(264, 300);
        var box = new VBoxContainer();
        AddChild(box);

        var title = new Label { Text = Tr("Necropolis HR") };
        title.AddThemeFontSizeOverride("font_size", 20);
        box.AddChild(title);

        AddRow(box, UpgradeId.MarcherRate, Tr("Hire Marchers"), Tr("+0.25/s"));
        AddRow(box, UpgradeId.ShieldbearerRate, Tr("Hire Shieldbearers"), Tr("+0.1/s"));
        AddRow(box, UpgradeId.Recovery, Tr("Death Benefits"), Tr("+15% bounty"));
    }

    private void AddRow(VBoxContainer box, UpgradeId id, string name, string effect)
    {
        var btn = new Button { CustomMinimumSize = new Vector2(248, 56) };
        btn.Pressed += () => _sim.TryPurchase(id);
        box.AddChild(btn);
        _rows.Add((id, name, effect, btn));
    }

    public void Refresh()
    {
        foreach (var (id, name, effect, btn) in _rows)
        {
            int lvl = _sim.State.UpgradeLevels.GetValueOrDefault(id);
            double cost = _sim.GetUpgradeCost(id);
            btn.Text = $"{name} Lv{lvl} ({effect})\n{Fmt.Num(cost)} {Tr("bones")}";
            btn.Disabled = _sim.State.Bones < cost;
        }
    }
}
