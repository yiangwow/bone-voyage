```csharp
using System;

namespace BoneVoyage.Sim;

/// <summary>Result of replaying the deterministic sim over time that passed
/// while the game was closed (or the OS slept).</summary>
public sealed record OfflineReport(
    double ElapsedSeconds,    // wall-clock seconds since the save timestamp, floored at 0
    double SimulatedSeconds,  // seconds actually replayed after the cap
    bool Capped,              // true when ElapsedSeconds exceeded CapSeconds
    long Recycled,            // skeletons that died (were recycled) during the replay
    double BonesEarned,
    double GoldEarned,
    int KingdomsFell);

public static class OfflineCalculator
{
    public const double CapSeconds = 24.0 * 3600.0;

    /// <summary>Advances <paramref name="sim"/> in place by running the same
    /// deterministic ticks the live game would have run, aggregating what
    /// happened. Events are consumed here and never reach the view: offline
    /// progress is a report, not a replay.</summary>
    public static OfflineReport Run(Simulation sim, double elapsedSeconds)
    {
        double elapsed = Math.Max(0.0, elapsedSeconds);   // clock-skew guard
        bool capped = elapsed > CapSeconds;
        double simulated = capped ? CapSeconds : elapsed;
        long ticks = (long)(simulated / sim.Balance.TickSeconds);

        double bones0 = sim.State.Bones, gold0 = sim.State.Gold;
        long recycled = 0;
        int fell = 0;
        for (long i = 0; i < ticks; i++)
            foreach (var e in sim.Tick())
                switch (e)
                {
                    case SkeletonsDied d: recycled += d.Count; break;
                    case KingdomFell: fell++; break;
                }

        return new OfflineReport(elapsed, ticks * sim.Balance.TickSeconds, capped,
            recycled, sim.State.Bones - bones0, sim.State.Gold - gold0, fell);
    }
}
```

```csharp
using System.Globalization;
using System.Text;
using BoneVoyage.Sim;

namespace BoneVoyage.BalanceHarness;

/// <summary>Headless greedy-buyer progression harness. Replays the
/// deterministic sim for --hours of game time; every sim-second it repeatedly
/// buys the cheapest affordable upgrade (a maximally impatient shopkeeper).
/// Prints kingdom-fall timings and optionally writes a progression CSV so
/// pacing is tuned from data, not vibes.</summary>
public static class Program
{
    public static int Main(string[] args)
    {
        string balancePath = "game/balance.json";
        double hours = 20.0, csvEverySeconds = 10.0;
        string? csvPath = null;
        for (int i = 0; i < args.Length; i++)
            switch (args[i])
            {
                case "--hours": hours = double.Parse(args[++i], CultureInfo.InvariantCulture); break;
                case "--csv": csvPath = args[++i]; break;
                case "--interval": csvEverySeconds = double.Parse(args[++i], CultureInfo.InvariantCulture); break;
                default: balancePath = args[i]; break;
            }
        if (!File.Exists(balancePath))
        {
            Console.Error.WriteLine($"balance file not found: {balancePath}");
            return 1;
        }

        var balance = BalanceData.Load(File.ReadAllText(balancePath));
        var sim = new Simulation(balance);
        var upgrades = balance.Upgrades.Keys.ToArray();

        var csv = new StringBuilder("t_seconds,bones,gold,total_falls,cycle_tier,kingdom_index,employees\n");
        var falls = new List<(double T, KingdomFell E)>();

        long ticksPerSecond = (long)Math.Round(1.0 / balance.TickSeconds);
        long csvEveryTicks = Math.Max(1, (long)Math.Round(csvEverySeconds / balance.TickSeconds));
        long totalTicks = (long)(hours * 3600.0 / balance.TickSeconds);

        for (long tick = 1; tick <= totalTicks; tick++)
        {
            foreach (var e in sim.Tick())
                if (e is KingdomFell f)
                {
                    double t = tick * balance.TickSeconds;
                    falls.Add((t, f));
                    Console.WriteLine($"fall #{falls.Count} t={t:0}s ({t / 60.0:0.0} min)  " +
                        $"{balance.Kingdoms[f.KingdomIndex].Name} tier {f.CycleTier}");
                }

            if (tick % ticksPerSecond == 0) BuyGreedily(sim, upgrades);
            if (tick % csvEveryTicks == 0) AppendRow(csv, sim, tick * balance.TickSeconds);
        }

        Console.WriteLine($"summary: {falls.Count} falls in {hours}h; " +
            $"final bones={sim.State.Bones:0}, gold={sim.State.Gold:0}, tier={sim.State.CycleTier}");
        if (csvPath != null)
        {
            File.WriteAllText(csvPath, csv.ToString());
            Console.WriteLine($"csv: {csvPath}");
        }
        return 0;
    }

    private static void BuyGreedily(Simulation sim, UpgradeId[] upgrades)
    {
        while (true)
        {
            UpgradeId? best = null;
            double bestCost = double.MaxValue;
            foreach (var id in upgrades)
            {
                double cost = sim.GetUpgradeCost(id);
                double wallet = sim.Balance.Upgrades[id].Currency == Currency.Gold
                    ? sim.State.Gold : sim.State.Bones;
                if (cost <= wallet && cost < bestCost) { best = id; bestCost = cost; }
            }
            if (best == null) return;
            sim.TryPurchase(best.Value);
        }
    }

    private static void AppendRow(StringBuilder csv, Simulation sim, double t)
    {
        int alive = 0;
        for (int ty = 0; ty < sim.State.Counts.GetLength(0); ty++)
            for (int z = 0; z < sim.State.Counts.GetLength(1); z++)
                alive += sim.State.Counts[ty, z];
        csv.Append(string.Create(CultureInfo.InvariantCulture,
            $"{t:0},{sim.State.Bones:0.##},{sim.State.Gold:0.##},{sim.State.TotalFalls}," +
            $"{sim.State.CycleTier},{sim.State.KingdomIndex},{alive}\n"));
    }
}
```

```python
"""Chiptune-adjacent SFX + ambient loop, synthesized from code (sprites' audio
sibling). stdlib only. 22050 Hz mono 16-bit WAVs into game/assets/audio/.
Deterministic: fixed seed, so regeneration is diff-clean."""
import math, os, random, struct, wave

SR = 22050
OUT = os.path.join(os.path.dirname(__file__), "..", "..", "game", "assets", "audio")
random.seed(7)


def write(name, samples):
    os.makedirs(OUT, exist_ok=True)
    path = os.path.join(OUT, name)
    with wave.open(path, "w") as w:
        w.setnchannels(1); w.setsampwidth(2); w.setframerate(SR)
        w.writeframes(b"".join(
            struct.pack("<h", max(-32767, min(32767, int(s * 32767)))) for s in samples))
    print(f"wrote {name}: {len(samples) / SR:.2f}s")


def env(i, n, a, r):
    """linear attack over a seconds, linear release over the final r seconds"""
    t, dur = i / SR, n / SR
    g = min(1.0, t / a) if a > 0 else 1.0
    rel = min(1.0, (dur - t) / r) if r > 0 else 1.0
    return g * max(0.0, rel)


def square(f, t):
    return 0.6 if math.sin(2 * math.pi * f * t) >= 0 else -0.6


def tri(f, t):
    p = (t * f) % 1.0
    return 4 * abs(p - 0.5) - 1.0


def note(freq, dur, wavefn=square, vol=0.4, a=0.004, r=0.05):
    n = int(dur * SR)
    return [wavefn(freq, i / SR) * env(i, n, a, r) * vol for i in range(n)]


def mix(*tracks):
    n = max(len(t) for t in tracks)
    return [sum(t[i] for t in tracks if i < len(t)) for i in range(n)]


# spawn: bright little C6->E6 punch-in ding
write("sfx_spawn.wav", note(1047, 0.07, vol=0.3) + note(1319, 0.10, vol=0.3))

# death: cheerful descending E6 C6 G5 triangle chime
write("sfx_death.wav",
      note(1319, 0.08, tri, 0.32) + note(1047, 0.08, tri, 0.30) + note(784, 0.11, tri, 0.28))

# whack: noise thud + low square knock
thud_n = int(0.12 * SR)
noise = [random.uniform(-1, 1) * env(i, thud_n, 0.002, 0.09) * 0.35 for i in range(thud_n)]
write("sfx_whack.wav", mix(noise, note(90, 0.12, square, 0.30, a=0.002, r=0.08)))

# fanfare: C5 E5 G5 C6 arpeggio + held triad
arp = sum((note(f, 0.12, square, 0.30) for f in (523, 659, 784, 1047)), [])
chord = mix(note(523, 0.6, tri, 0.22, r=0.25), note(659, 0.6, tri, 0.22, r=0.25),
            note(1047, 0.6, tri, 0.22, r=0.25))
write("sfx_fanfare.wav", arp + chord)

# ambient: 12 s pad. Frequencies chosen so freq * 12 is an integer (whole
# cycles per loop) and LFOs run whole cycles too -> seamless, click-free loop.
LOOP = 12.0
n = int(LOOP * SR)
pad = []
voices = [(130.0, 2), (195.0, 3), (325.0, 5)]  # ~C3, ~G3, ~E4; LFO cycles per loop
for i in range(n):
    t = i / SR
    s = 0.0
    for freq, lfo_cycles in voices:
        lfo = 0.75 + 0.25 * math.cos(2 * math.pi * lfo_cycles * t / LOOP)
        s += math.sin(2 * math.pi * freq * t) * lfo
    pad.append(s * 0.16 / len(voices) * 3)  # keep it a whisper under the SFX
write("ambient_loop.wav", pad)
```

```csharp
using System;
using System.Collections.Generic;
using BoneVoyage.Sim;
using Godot;

namespace BoneVoyage.Game;

/// <summary>All playback lives here: pooled SFX players + one looping ambient
/// player. Sim events map to SFX with per-sound rate limits so 50 deaths in a
/// tick play once, not fifty times.</summary>
public partial class AudioManager : Node
{
    private const int PoolSize = 8;
    private readonly List<AudioStreamPlayer> _pool = new();
    private readonly Dictionary<string, AudioStream> _streams = new();
    private readonly Dictionary<string, ulong> _lastMs = new();
    private static readonly Dictionary<string, ulong> MinGapMs = new()
    { ["spawn"] = 350, ["death"] = 130, ["whack"] = 160, ["fanfare"] = 1500 };
    private AudioStreamPlayer _music = null!;

    public double MusicVolume { get; private set; }
    public double SfxVolume { get; private set; }

    public AudioManager(SaveSettings settings)
    {
        MusicVolume = settings.MusicVolume;
        SfxVolume = settings.SfxVolume;
    }

    public override void _Ready()
    {
        foreach (var name in new[] { "spawn", "death", "whack", "fanfare" })
            _streams[name] = GD.Load<AudioStream>($"res://assets/audio/sfx_{name}.wav");
        for (int i = 0; i < PoolSize; i++)
        {
            var p = new AudioStreamPlayer();
            AddChild(p);
            _pool.Add(p);
        }

        var ambient = GD.Load<AudioStreamWav>("res://assets/audio/ambient_loop.wav");
        ambient.LoopMode = AudioStreamWav.LoopModeEnum.Forward;
        ambient.LoopEnd = ambient.Data.Length / 2;   // 16-bit mono: 2 bytes per frame
        _music = new AudioStreamPlayer { Stream = ambient };
        AddChild(_music);
        ApplyVolumes();
        _music.Play();
    }

    public void SetVolumes(double music, double sfx)
    {
        MusicVolume = Math.Clamp(music, 0.0, 1.0);
        SfxVolume = Math.Clamp(sfx, 0.0, 1.0);
        ApplyVolumes();
    }

    public SaveSettings Snapshot() => new(MusicVolume, SfxVolume);

    public void OnSimEvents(List<ISimEvent> events)
    {
        foreach (var e in events)
            switch (e)
            {
                case SkeletonsSpawned: Play("spawn"); break;
                case SkeletonsDied: Play("death"); break;
                case WallHit: Play("whack"); break;
                case KingdomFell: Play("fanfare"); break;
            }
    }

    private void ApplyVolumes()
    {
        _music.VolumeDb = LinearToDb(MusicVolume);
        foreach (var p in _pool) p.VolumeDb = LinearToDb(SfxVolume);
    }

    private static float LinearToDb(double v) =>
        v <= 0.001 ? -80f : Mathf.LinearToDb((float)v);

    private void Play(string name)
    {
        ulong now = Time.GetTicksMsec();
        if (_lastMs.TryGetValue(name, out ulong last) && now - last < MinGapMs[name]) return;
        foreach (var p in _pool)
        {
            if (p.Playing) continue;
            _lastMs[name] = now;
            p.Stream = _streams[name];
            p.PitchScale = 0.94f + (float)GD.RandRange(0.0, 0.12);  // cosmetic variety
            p.Play();
            return;
        }
    }
}
```
